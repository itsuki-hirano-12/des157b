Triangle drawing function

The function draws triangles on console. It's from the first JS experiment in 157A rewritten as a function.

Named Function

            function drawTriangles(height, waves) {
                console.log(triangle);
                
                for ( let i=0; i < waves; i++) {
                    for (let j = 0; j < height - 1; j++) {
                        triangle += "#";
                        console.log(triangle);
                    }
                    for (let j = 0; j < height - 1; j++) {
                        triangle = triangle.slice(1);
                        console.log(triangle);
                    }
                }
            }
